cmake_minimum_required(VERSION 3.28)
project(sage_unified VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(SAGE_ENABLE_CUDA "Build the NInfer-derived SM75 kernel library" OFF)
option(SAGE_BUILD_STRATA "Build the patched, single-owner Strata runtime" OFF)
set(SAGE_STRATA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/PolyStrata" CACHE PATH "Pinned and patched Strata checkout")

add_library(sage_scheduler STATIC src/slot_pool.cpp)
target_include_directories(sage_scheduler PUBLIC include)
if(BUILD_TESTING)
  add_executable(slot_pool_test tests/slot_pool_test.cpp)
  target_link_libraries(slot_pool_test PRIVATE sage_scheduler)
  add_test(NAME slot_pool COMMAND slot_pool_test)
  add_executable(strata_policy_test tests/strata_policy_test.cpp)
  target_include_directories(strata_policy_test PRIVATE integration/strata/include)
  add_test(NAME strata_policy COMMAND strata_policy_test)
  add_executable(strata_expert_q4k_test tests/strata_expert_q4k_test.cpp)
  target_include_directories(strata_expert_q4k_test PRIVATE integration/strata/include kernels/include)
  add_test(NAME strata_expert_q4k COMMAND strata_expert_q4k_test)
  add_executable(q4k_oracle_test kernels/tests/q4k_slots_parity.cpp)
  target_include_directories(q4k_oracle_test PRIVATE kernels/include)
  target_compile_definitions(q4k_oracle_test PRIVATE SAGE_Q4K_CPU_ORACLE_ONLY=1)
  add_test(NAME q4k_oracle_fixture COMMAND q4k_oracle_test)
  add_executable(token_file_test tests/token_file_test.cpp)
  target_include_directories(token_file_test PRIVATE integration/strata/include)
  add_test(NAME token_file COMMAND token_file_test)
endif()

if(SAGE_ENABLE_CUDA)
  set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "Target architecture")
  if(NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "75")
    message(FATAL_ERROR "This product targets SM75; use CMAKE_CUDA_ARCHITECTURES=75")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_subdirectory(kernels)
endif()

if(SAGE_BUILD_STRATA)
  if(NOT SAGE_ENABLE_CUDA)
    message(FATAL_ERROR "SAGE_BUILD_STRATA requires SAGE_ENABLE_CUDA=ON")
  endif()
  if(NOT EXISTS "${SAGE_STRATA_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Run python tools/bootstrap.py first")
  endif()
  file(READ "${SAGE_STRATA_SOURCE_DIR}/CMakeLists.txt" _strata_cmake)
  if(NOT _strata_cmake MATCHES "STRATA_SAGE_INTEGRATION_DIR")
    message(FATAL_ERROR "Strata integration patch missing; run tools/bootstrap.py")
  endif()
  set(STRATA_ENABLE_CUDA ON CACHE BOOL "" FORCE)
  set(STRATA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(STRATA_NATIVE_EXPERTS ON CACHE BOOL "" FORCE)
  set(STRATA_SAGE_INTEGRATION_DIR "${CMAKE_CURRENT_SOURCE_DIR}/integration/strata" CACHE PATH "" FORCE)
  add_subdirectory("${SAGE_STRATA_SOURCE_DIR}" strata)
  if(NOT TARGET strata-poly)
    message(FATAL_ERROR "Pinned configuration did not create strata-poly")
  endif()
endif()
